Nodemailer
const nodemailer = require("nodemailer");

const t = nodemailer.createTransport({
  host: "smtp.geserd.com",
  port: 465,
  secure: true,
  auth: { user: "geserd", pass: "gs_live_..." }
});

await t.sendMail({
  from: "hi@yourdomain.com",
  to: "ada@example.com",
  subject: "Hello"
});
